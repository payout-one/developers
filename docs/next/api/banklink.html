POST

List Accounts

View as MarkdownSpec
/accounts

Retrieve current user connected accounts

Parameters

pageSizequeryNumber of accounts per page, default is 50, maximum is 100
pagequeryIndex of page to display, starting from 1
orderquerySort order: * `asc` - Ascending, from A to Z * `desc` - Descending, from Z to A default is `asc`
Request
curl -X POST https://app.payout.one/accounts \
  -H "Authorization: Bearer $TOKEN"
const res = await fetch("https://app.payout.one/accounts", {
  method: "POST",
  headers: {
    Authorization: `Bearer ${process.env.PAYOUT_TOKEN}`,
  },
});
const data = await res.json();
import os, requests

res = requests.post(
    "https://app.payout.one/accounts",
    headers={"Authorization": f"Bearer {os.environ['PAYOUT_TOKEN']}"},
)
data = res.json()
<?php
$ch = curl_init();
curl_setopt($ch, CURLOPT_URL, "https://app.payout.one/accounts");
curl_setopt($ch, CURLOPT_RETURNTRANSFER, true);
curl_setopt($ch, CURLOPT_CUSTOMREQUEST, "POST");
curl_setopt($ch, CURLOPT_HTTPHEADER, ["Authorization: Bearer " . getenv("PAYOUT_TOKEN")]);
$data = json_decode(curl_exec($ch), true);
curl_close($ch);

Account Balances

View as MarkdownSpec
/accounts/balance

Retrieve current user account balance directly from bank

Request
curl -X POST https://app.payout.one/accounts/balance \
  -H "Authorization: Bearer $TOKEN" \
  -H "Content-Type: application/json" \
  -d '{
       "bank": "vub",
       "iban": "nostrud incididunt aute occaecat"
     }'
const res = await fetch("https://app.payout.one/accounts/balance", {
  method: "POST",
  headers: {
    Authorization: `Bearer ${process.env.PAYOUT_TOKEN}`,
    "Content-Type": "application/json",
  },
  body: JSON.stringify({
    "bank": "vub",
    "iban": "nostrud incididunt aute occaecat"
  }),
});
const data = await res.json();
import os, requests

res = requests.post(
    "https://app.payout.one/accounts/balance",
    headers={"Authorization": f"Bearer {os.environ['PAYOUT_TOKEN']}"},
    json={
        "bank": "vub",
        "iban": "nostrud incididunt aute occaecat"
    },
)
data = res.json()
<?php
$ch = curl_init();
curl_setopt($ch, CURLOPT_URL, "https://app.payout.one/accounts/balance");
curl_setopt($ch, CURLOPT_RETURNTRANSFER, true);
curl_setopt($ch, CURLOPT_CUSTOMREQUEST, "POST");
