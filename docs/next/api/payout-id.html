PayoutID OAuth2

Payout OAuth2 authentication and authorization server

GET

Authorization redirect for user

View as MarkdownSpec
/oauth/authorize

Endpoint where client should redirect user for authorization

Parameters

client_idquery(Required) Client ID of application that is triing to authorize user
response_typequery(Required)
redirect_uriquery(Required) Redirect uri that was registered for this client
scopequery(Required) Required scopes to authorize
code_challengequerycode_challenge code_challenge = BASE64URL-ENCODE(SHA256(ASCII(code_verifier))) see RFC 7636 https://tools.ietf.org/html/rfc7636
code_challenge_methodqueryCode challenge method, must by set to `S256`
statequeryThis parameter is send back in redirect url to indetifiy request/response pairs
Request
curl -X GET https://app.payout.one/oauth/authorize \
  -H "Authorization: Bearer $TOKEN"
const res = await fetch("https://app.payout.one/oauth/authorize", {
  method: "GET",
  headers: {
    Authorization: `Bearer ${process.env.PAYOUT_TOKEN}`,
  },
});
const data = await res.json();
import os, requests

res = requests.get(
    "https://app.payout.one/oauth/authorize",
    headers={"Authorization": f"Bearer {os.environ['PAYOUT_TOKEN']}"},
)
data = res.json()
<?php
$ch = curl_init();
curl_setopt($ch, CURLOPT_URL, "https://app.payout.one/oauth/authorize");
curl_setopt($ch, CURLOPT_RETURNTRANSFER, true);
curl_setopt($ch, CURLOPT_CUSTOMREQUEST, "GET");
curl_setopt($ch, CURLOPT_HTTPHEADER, ["Authorization: Bearer " . getenv("PAYOUT_TOKEN")]);
$data = json_decode(curl_exec($ch), true);
curl_close($ch);

Endpoint to retrieve authorization token

View as MarkdownSpec
/oauth/token

Endpoint where client can retrieve authorization token after successfull authorization or refresh expired access token with valid refresh token

Request
curl -X POST https://app.payout.one/oauth/token \
  -H "Authorization: Bearer $TOKEN" \
  -H "Content-Type: application/json" \
  -d '{
       "grant_type": "authorization_code",
       "client_id": "2591CF97-1AFB-4D63-8B42-2D0CD6194CAF",
       "client_secret": "4c4f4df3-d026--8c4b-98e0434c882a",
       "code": "8167B030-A72A-466E-9EDC-7377B15C9C7B",
       "redirect_uri": "https://example.com",
       "code_verifier": "60ac297b-c100-42ce-be0d-46b14a0b0hee"
     }'
const res = await fetch("https://app.payout.one/oauth/token", {
  method: "POST",
  headers: {
    Authorization: `Bearer ${process.env.PAYOUT_TOKEN}`,
    "Content-Type": "application/json",
  },
  body: JSON.stringify({
    "grant_type": "authorization_code",
    "client_id": "2591CF97-1AFB-4D63-8B42-2D0CD6194CAF",
    "client_secret": "4c4f4df3-d026--8c4b-98e0434c882a",
    "code": "8167B030-A72A-466E-9EDC-7377B15C9C7B",
    "redirect_uri": "https://example.com",
    "code_verifier": "60ac297b-c100-42ce-be0d-46b14a0b0hee"
  }),
});
const data = await res.json();
import os, requests

res = requests.post(
    "https://app.payout.one/oauth/token",
    headers={"Authorization": f"Bearer {os.environ['PAYOUT_TOKEN']}"},
    json={
        "grant_type": "authorization_code",
        "client_id": "2591CF97-1AFB-4D63-8B42-2D0CD6194CAF",
        "client_secret": "4c4f4df3-d026--8c4b-98e0434c882a",
        "code": "8167B030-A72A-466E-9EDC-7377B15C9C7B",
        "redirect_uri": "https://example.com",
        "code_verifier": "60ac297b-c100-42ce-be0d-46b14a0b0hee"
    },
)
data = res.json()
<?php
$ch = curl_init();
curl_setopt($ch, CURLOPT_URL, "https://app.payout.one/oauth/token");
curl_setopt($ch, CURLOPT_RETURNTRANSFER, true);
curl_setopt($ch, CURLOPT_CUSTOMREQUEST, "POST");
curl_setopt($ch, CURLOPT_POSTFIELDS, json_encode([
  "grant_type" => "authorization_code",
  "client_id" => "2591CF97-1AFB-4D63-8B42-2D0CD6194CAF",
  "client_secret" => "4c4f4df3-d026--8c4b-98e0434c882a",
  "code" => "8167B030-A72A-466E-9EDC-7377B15C9C7B",
  "redirect_uri" => "https://example.com",
  "code_verifier" => "60ac297b-c100-42ce-be0d-46b14a0b0hee"
]));
curl_setopt($ch, CURLOPT_HTTPHEADER, ["Authorization: Bearer " . getenv("PAYOUT_TOKEN"), "Content-Type: application/json"]);
$data = json_decode(curl_exec($ch), true);
curl_close($ch);
